# Day5 - 人机协同(Human-in-the-Loop)

> 今天我们让 Agent 学会"请示":在执行敏感操作前暂停,等人类批准或修改后再继续。这是把 Agent 从"Demo"推向"生产可信赖"的关键能力。

---

## 一、当日目标

1. 理解 Human-in-the-Loop(HITL)的典型场景:为什么需要人工介入、介入的几种形式。
2. 掌握 `interrupt` 函数的用法:在节点内部主动暂停并向人类请求输入。
3. 掌握 Tool 调用审批:用 `interrupt_before` 拦截工具执行,人工确认后再放行。
4. 理解 `Command(resume=...)` 恢复机制:把人工输入注入回图继续执行。
5. 实现一个"发邮件前需人工确认"的 Agent。

---

## 二、HITL 的典型场景

### 2.1 为什么需要人工介入

LLM 不可 100% 信任,以下场景需要人工把关:
- **高风险操作**:发邮件、转账、删除数据、调用付费 API。
- **模糊决策**:LLM 不确定时,把选择权交给人类。
- **合规要求**:某些行业规定关键决策必须有人工签字。
- **质量校验**:翻译、文案生成后,人工润色再发布。

### 2.2 LangGraph 的三种 HITL 机制

| 机制 | 触发位置 | 适用场景 |
|------|---------|---------|
| `interrupt_before` / `interrupt_after` | 节点边界 | 在指定节点前/后暂停(Tool 审批最常用) |
| `interrupt()` 函数 | 节点内部 | 节点运行中主动暂停,向人类请求输入 |
| 动态中断(条件 + interrupt) | 节点内部按条件 | 只有满足条件才暂停(如金额 > 1000 才审批) |

三种机制都依赖 Checkpointer(Day4)——暂停时状态被持久化,人类响应后从 checkpoint 恢复继续。

---

## 三、Tool 调用审批(interrupt_before)

最经典的 HITL:Agent 要调工具时暂停,人类看一眼 tool_calls,批准就继续,拒绝就改方向。

### 3.1 完整示例:工具审批

```python
"""Day5 Tool 调用审批示例"""
import os
import sqlite3
from typing import TypedDict, Annotated
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, ToolMessage
from langgraph.graph import StateGraph, START, END
from langgraph.graph.message import add_messages
from langgraph.checkpoint.sqlite import SqliteSaver

load_dotenv()


def send_email(to: str, subject: str, body: str) -> str:
    """发送邮件(模拟,实际会真正发出)"""
    return f"邮件已发送给 {to},主题:{subject}"


def get_weather(city: str) -> str:
    """查询天气"""
    return {"北京": "晴 25°C", "上海": "多云 28°C"}.get(city, "无数据")


tools = [send_email, get_weather]
tools_by_name = {t.__name__: t for t in tools}
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0).bind_tools(tools)


class AgentState(TypedDict):
    messages: Annotated[list, add_messages]


def agent_node(state):
    return {"messages": [llm.invoke(state["messages"])]}


def tool_node(state):
    last: AIMessage = state["messages"][-1]
    out = []
    for c in last.tool_calls:
        try:
            r = tools_by_name[c["name"]](**c["args"])
        except (KeyError, TypeError, ValueError) as e:
            r = f"错误:{type(e).__name__}"
        out.append(ToolMessage(content=str(r), tool_call_id=c["id"]))
    return {"messages": out}


def should_use_tool(state):
    last = state["messages"][-1]
    if hasattr(last, "tool_calls") and last.tool_calls:
        return "tools"
    return END


builder = StateGraph(AgentState)
builder.add_node("agent", agent_node)
builder.add_node("tools", tool_node)
builder.add_edge(START, "agent")
builder.add_conditional_edges("agent", should_use_tool)
builder.add_edge("tools", "agent")

conn = sqlite3.connect("hitl.db", check_same_thread=False)
# 关键:interrupt_before=["tools"] 让工具执行前暂停
graph = builder.compile(
    checkpointer=SqliteSaver(conn),
    interrupt_before=["tools"],
)


def run_with_approval(thread_id: str, user_input: str, approver_fn):
    """运行 Agent,遇到工具调用时调用 approver_fn 请求人工审批"""
    config = {"configurable": {"thread_id": thread_id}}
    graph.invoke({"messages": [HumanMessage(content=user_input)]}, config=config)

    # 循环处理:每次暂停在 tools 前,审批后继续
    while True:
        state = graph.get_state(config)
        if not state.next:
            # 已结束
            break
        # 暂停在 tools 前,审批
        last_msg: AIMessage = state.values["messages"][-1]
        print("\n=== 待审批的工具调用 ===")
        for c in last_msg.tool_calls:
            print(f"工具:{c['name']}, 参数:{c['args']}")

        approved = approver_fn(last_msg.tool_calls)
        if approved:
            print(">>> 审批通过,执行工具")
            graph.invoke(None, config=config)  # 继续执行 tools
        else:
            print(">>> 审批拒绝,注入拒绝消息")
            # 用 ToolMessage 告诉 LLM 工具被拒绝
            reject_msgs = [
                ToolMessage(
                    content="人工拒绝了该工具调用,请不要执行。",
                    tool_call_id=c["id"],
                )
                for c in last_msg.tool_calls
            ]
            graph.invoke({"messages": reject_msgs}, config=config)

    final = graph.get_state(config)
    return final.values["messages"][-1].content


# 模拟人工审批函数
def human_approve(tool_calls) -> bool:
    """模拟人工审批逻辑"""
    for c in tool_calls:
        if c["name"] == "send_email":
            # 发邮件必须人工确认
            answer = input(f"允许发送邮件给 {c['args'].get('to')}?(yes/no): ")
            return answer.strip().lower() == "yes"
    # 非敏感工具直接放行
    return True


if __name__ == "__main__":
    tid = "hitl-demo-1"
    reply = run_with_approval(
        tid,
        "帮我给 boss@company.com 发一封邮件,主题'请假申请',正文'我想请假三天'",
        human_approve,
    )
    print("\n最终回复:", reply)
```

### 3.2 流程解析

1. 用户要求发邮件 → Agent 调用 `send_email` 工具。
2. 图在 `tools` 节点前暂停(`interrupt_before=["tools"]`)。
3. `get_state` 看到 `state.next == ("tools",)`,取出待执行的 tool_calls。
4. 调用 `human_approve`:如果是 `send_email`,弹 input 让人确认。
5. 批准:`invoke(None)` 继续执行 tools,邮件发出。
6. 拒绝:`invoke({"messages": [拒绝 ToolMessage]})` 注入拒绝信息,Agent 会看到"工具被拒",重新组织回复。

### 3.3 关键点

- 暂停后必须用 `invoke(None, config)` 继续,不能用 `invoke(新输入)`(会追加新消息而不是恢复)。
- 拒绝时通过 `ToolMessage` 反馈给 LLM,这是符合 OpenAI function calling 协议的做法——每个 tool_call 必须有对应的 ToolMessage 响应,否则 LLM 会困惑。
- 这种"审批"机制对所有工具生效。若只想审批特定工具,见第五节的动态中断。

## 四、interrupt() 函数:节点内主动请求人类输入

`interrupt` 比 `interrupt_before` 更精细:它让你在节点**执行过程中**主动暂停,并可以向人类请求一个具体的输入值。

### 4.1 基本用法

```python
from langgraph.types import interrupt, Command

def review_node(state):
    # 节点执行到一半,向人类请求确认
    user_feedback = interrupt({
        "question": "这段文案是否 OK?",
        "content": state["draft"],
    })
    # 人类反馈会作为 interrupt() 的返回值
    if user_feedback == "approve":
        return {"status": "approved"}
    return {"status": "revised", "draft": user_feedback}
```

调用方恢复执行时,用 `Command(resume=...)` 把人类输入注入:

```python
from langgraph.types import Command

# 第一次调用,会在 interrupt 处暂停
graph.invoke({"draft": "初稿文案"}, config=config)

# 人类给出反馈,恢复执行
result = graph.invoke(
    Command(resume="approve"),
    config=config,
)
```

### 4.2 完整示例:文案审核

```python
"""Day5 interrupt() 文案审核示例"""
import os
import sqlite3
from typing import TypedDict
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.sqlite import SqliteSaver
from langgraph.types import interrupt, Command

load_dotenv()


class DraftState(TypedDict):
    topic: str
    draft: str
    final: str


def write_draft(state: DraftState) -> dict:
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0.7)
    prompt = f"为以下主题写一句营销文案,20 字以内:{state['topic']}"
    draft = llm.invoke(prompt).content
    return {"draft": draft}


def review_node(state: DraftState) -> Command:
    # 主动暂停,向人类请求审核结果
    feedback = interrupt({
        "prompt": "请审核以下文案",
        "draft": state["draft"],
        "options": "输入 'ok' 通过,或输入修改后的文案",
    })
    if feedback.strip().lower() in ("ok", "通过", "approve"):
        return Command(update={"final": state["draft"]}, goto="publish")
    # 人类给了修改稿
    return Command(update={"final": feedback.strip()}, goto="publish")


def publish(state: DraftState) -> dict:
    return {"final": state["final"]}  # 实际可写入数据库


builder = StateGraph(DraftState)
builder.add_node("write", write_draft)
builder.add_node("review", review_node)
builder.add_node("publish", publish)
builder.add_edge(START, "write")
# write 到 review 用普通边(interrupt 在 review 内部触发)
builder.add_edge("write", "review")
builder.add_edge("publish", END)

conn = sqlite3.connect("interrupt_demo.db", check_same_thread=False)
graph = builder.compile(checkpointer=SqliteSaver(conn))


def main():
    config = {"configurable": {"thread_id": "draft-1"}}
    # 第一次:写初稿,然后在 review 节点 interrupt 暂停
    graph.invoke({"topic": "新款手机发布", "draft": "", "final": ""}, config=config)

    state = graph.get_state(config)
    print("=== 待审核文案 ===")
    print(state.values.get("draft"))

    # 模拟人类审核
    human_input = input("你的审核意见(ok 或修改稿): ")

    # 用 Command(resume=...) 恢复,interrupt() 的返回值就是 human_input
    result = graph.invoke(Command(resume=human_input), config=config)
    print("\n=== 最终发布 ===")
    print(result["final"])


if __name__ == "__main__":
    main()
```

### 4.3 interrupt vs interrupt_before 对比

| 特性 | `interrupt_before` | `interrupt()` 函数 |
|------|--------------------|--------------------|
| 触发位置 | 节点边界(进入节点前) | 节点内部任意位置 |
| 暂停时状态 | 节点尚未执行 | 节点已执行到 interrupt 处 |
| 恢复方式 | `invoke(None)` | `invoke(Command(resume=值))` |
| 能否传人类输入 | 否(只是放行) | 是(resume 值成为 interrupt 返回值) |
| 灵活性 | 低(节点级) | 高(代码行级) |

新版本推荐用 `interrupt()`,它更精细、表达力更强。

## 五、动态中断(条件性审批)

不是所有工具调用都需要审批——只有满足条件(如金额超过阈值)才暂停。这通过在节点内部结合条件判断和 `interrupt` 实现。

### 5.1 示例:大额转账审批

```python
"""Day5 动态中断:大额转账才审批"""
import os
import sqlite3
from typing import TypedDict, Annotated
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, ToolMessage
from langgraph.graph import StateGraph, START, END
from langgraph.graph.message import add_messages
from langgraph.checkpoint.sqlite import SqliteSaver
from langgraph.types import interrupt, Command

load_dotenv()

THRESHOLD = 1000  # 超过 1000 元需要审批


def transfer(to: str, amount: float) -> str:
    """转账(模拟)"""
    return f"已向 {to} 转账 {amount} 元"


tools = [transfer]
tools_by_name = {t.__name__: t for t in tools}
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0).bind_tools(tools)


class AgentState(TypedDict):
    messages: Annotated[list, add_messages]


def agent_node(state):
    return {"messages": [llm.invoke(state["messages"])]}


def smart_tool_node(state):
    """智能工具节点:大额才审批"""
    last: AIMessage = state["messages"][-1]
    out = []
    for c in last.tool_calls:
        if c["name"] == "transfer":
            amount = float(c["args"].get("amount", 0))
            if amount > THRESHOLD:
                # 动态中断:请求人工审批
                decision = interrupt({
                    "type": "transfer_approval",
                    "to": c["args"]["to"],
                    "amount": amount,
                    "message": f"转账 {amount} 元给 {c['args']['to']},是否允许?(yes/no)",
                })
                if decision.strip().lower() not in ("yes", "ok", "允许"):
                    out.append(ToolMessage(
                        content=f"人工拒绝了转账 {amount} 元给 {c['args']['to']}",
                        tool_call_id=c["id"],
                    ))
                    continue
        # 执行工具
        try:
            r = tools_by_name[c["name"]](**c["args"])
        except (KeyError, TypeError, ValueError) as e:
            r = f"错误:{type(e).__name__}"
        out.append(ToolMessage(content=str(r), tool_call_id=c["id"]))
    return {"messages": out}


def should_use_tool(state):
    last = state["messages"][-1]
    if hasattr(last, "tool_calls") and last.tool_calls:
        return "tools"
    return END


builder = StateGraph(AgentState)
builder.add_node("agent", agent_node)
builder.add_node("tools", smart_tool_node)
builder.add_edge(START, "agent")
builder.add_conditional_edges("agent", should_use_tool)
builder.add_edge("tools", "agent")

conn = sqlite3.connect("dynamic_hitl.db", check_same_thread=False)
graph = builder.compile(checkpointer=SqliteSaver(conn))


def main():
    config = {"configurable": {"thread_id": "pay-1"}}

    # 小额:不审批,直接转
    print("--- 小额转账(500 元)---")
    graph.invoke({"messages": [HumanMessage(content="给 alice 转 500 元")]}, config=config)
    state = graph.get_state(config)
    print("结果:", state.values["messages"][-1].content)

    # 大额:会触发 interrupt
    print("\n--- 大额转账(2000 元,需审批)---")
    config2 = {"configurable": {"thread_id": "pay-2"}}
    graph.invoke({"messages": [HumanMessage(content="给 bob 转 2000 元")]}, config=config2)
    state2 = graph.get_state(config2)
    # 此时在 interrupt 处暂停
    print("待审批:", state2.values.get("messages", [])[-1].content if state2.next else "已结束")

    if state2.next:  # 仍在暂停状态
        # 模拟人工审批
        decision = input("是否允许这笔 2000 元转账?(yes/no): ")
        result = graph.invoke(Command(resume=decision), config=config2)
        print("最终:", result["messages"][-1].content)


if __name__ == "__main__":
    main()
```

### 5.2 动态中断的优势

- **精细控制**:不同工具、不同参数,审批策略不同。
- **业务规则内嵌**:阈值、白名单等逻辑直接写在节点里。
- **用户体验好**:小额免审,大额才打断,不打扰日常操作。

## 六、恢复执行的两种方式对比

| 恢复方式 | 适用场景 | 代码 |
|---------|---------|------|
| `invoke(None, config)` | `interrupt_before` 暂停后,直接放行 | 不传新输入 |
| `invoke(Command(resume=值), config)` | `interrupt()` 暂停后,传人类输入 | resume 值成为 interrupt 返回值 |
| `invoke({"messages": [...]}, config)` | 追加新消息(如注入拒绝 ToolMessage) | 普通状态更新 |

混淆这三种是常见错误。记住:
- `interrupt_before` → `invoke(None)`。
- `interrupt()` → `invoke(Command(resume=...))`。
- 想追加消息 → `invoke({"messages": [...]})`。

## 七、实战:带审批的邮件 Agent

整合本日所学,做一个"发邮件需审批、查天气免审批"的 Agent:

```python
"""Day5 实战:带审批的邮件 Agent"""
import os
import sqlite3
from typing import TypedDict, Annotated
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, ToolMessage
from langgraph.graph import StateGraph, START, END
from langgraph.graph.message import add_messages
from langgraph.checkpoint.sqlite import SqliteSaver
from langgraph.types import interrupt, Command

load_dotenv()


def send_email(to: str, subject: str, body: str) -> str:
    """发送邮件(模拟)"""
    return f"已发送:{to} | {subject} | {body[:30]}..."


def get_weather(city: str) -> str:
    """查询天气"""
    return {"北京": "晴 25°C", "上海": "多云 28°C"}.get(city, "无数据")


SENSITIVE_TOOLS = {"send_email"}
tools = [send_email, get_weather]
tools_by_name = {t.__name__: t for t in tools}
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0).bind_tools(tools)


class AgentState(TypedDict):
    messages: Annotated[list, add_messages]


def agent_node(state):
    return {"messages": [llm.invoke(state["messages"])]}


def guarded_tool_node(state):
    """敏感工具需人工审批,普通工具直接执行"""
    last: AIMessage = state["messages"][-1]
    out = []
    pending_sensitive = []
    for c in last.tool_calls:
        if c["name"] in SENSITIVE_TOOLS:
            pending_sensitive.append(c)
        else:
            try:
                r = tools_by_name[c["name"]](**c["args"])
            except (KeyError, TypeError, ValueError) as e:
                r = f"错误:{type(e).__name__}"
            out.append(ToolMessage(content=str(r), tool_call_id=c["id"]))

    if pending_sensitive:
        # 把敏感工具的详情发给人类审批
        details = [
            {"name": c["name"], "args": c["args"], "id": c["id"]}
            for c in pending_sensitive
        ]
        decision = interrupt({"type": "sensitive_approval", "calls": details})
        # decision 是 dict: {call_id: "approve"|"reject"}
        for c in pending_sensitive:
            verdict = decision.get(c["id"], "reject")
            if verdict == "approve":
                try:
                    r = tools_by_name[c["name"]](**c["args"])
                except (KeyError, TypeError, ValueError) as e:
                    r = f"错误:{type(e).__name__}"
                out.append(ToolMessage(content=str(r), tool_call_id=c["id"]))
            else:
                out.append(ToolMessage(
                    content="人工拒绝执行该操作。",
                    tool_call_id=c["id"],
                ))
    return {"messages": out}


def should_use_tool(state):
    last = state["messages"][-1]
    if hasattr(last, "tool_calls") and last.tool_calls:
        return "tools"
    return END


builder = StateGraph(AgentState)
builder.add_node("agent", agent_node)
builder.add_node("tools", guarded_tool_node)
builder.add_edge(START, "agent")
builder.add_conditional_edges("agent", should_use_tool)
builder.add_edge("tools", "agent")

conn = sqlite3.connect("email_agent.db", check_same_thread=False)
graph = builder.compile(checkpointer=SqliteSaver(conn))


def chat_with_hitl(thread_id: str, user_input: str, approver_fn):
    config = {"configurable": {"thread_id": thread_id}}
    graph.invoke({"messages": [HumanMessage(content=user_input)]}, config=config)

    # 处理可能的 interrupt 循环
    while True:
        state = graph.get_state(config)
        if not state.next:
            break
        # 检查是否有待审批任务
        tasks = state.tasks or []
        interrupt_info = None
        for task in tasks:
            if hasattr(task, "interrupts") and task.interrupts:
                interrupt_info = task.interrupts[0].value
                break
        if interrupt_info and interrupt_info.get("type") == "sensitive_approval":
            print("\n=== 待审批 ===")
            for c in interrupt_info["calls"]:
                print(f"  {c['name']}({c['args']})")
            decision = approver_fn(interrupt_info["calls"])
            graph.invoke(Command(resume=decision), config=config)
        else:
            graph.invoke(None, config=config)

    final = graph.get_state(config)
    return final.values["messages"][-1].content


def auto_approver(calls) -> dict:
    """自动审批器:演示用,实际换成人工 UI"""
    result = {}
    for c in calls:
        ans = input(f"允许 {c['name']}({c['args']})? (yes/no): ")
        result[c["id"]] = "approve" if ans.strip().lower() in ("yes", "y") else "reject"
    return result


if __name__ == "__main__":
    tid = "email-1"
    # 这个会触发审批
    reply = chat_with_hitl(
        tid,
        "帮我给 team@x.com 发邮件主题'周报',正文'本周完成 LangGraph 学习'",
        auto_approver,
    )
    print("回复:", reply)

    # 这个不会触发审批
    tid2 = "weather-1"
    reply = chat_with_hitl(tid2, "北京天气怎么样?", auto_approver)
    print("回复:", reply)
```

## 八、当日小结

- HITL 让 Agent 在敏感操作前暂停等人类确认,是生产可信赖的关键。
- 三种机制:
  - `interrupt_before=["node"]`:节点边界暂停,`invoke(None)` 继续。
  - `interrupt()`:节点内部暂停,`invoke(Command(resume=值))` 继续,resume 值成为 interrupt 返回值。
  - 动态中断:节点内按条件调 `interrupt`,实现"只有特定情况才审批"。
- 拒绝工具时返回 `ToolMessage` 告知 LLM,符合 function calling 协议。
- 所有 HITL 都依赖 Checkpointer,暂停时状态持久化,恢复时从 checkpoint 加载。
- 检测 interrupt:查看 `state.tasks` 中的 `interrupts` 字段。

## 十、练习题答案

### 1. 答案:B
解析:`interrupt()` 暂停后,必须用 `Command(resume=值)` 恢复。`resume` 的值会成为 `interrupt()` 调用的返回值,节点代码继续往下执行。

### 2. 答案:A
解析:`interrupt_before` 是节点边界暂停,没有"人类输入"的概念,只需放行即可,用 `invoke(None, config)`。

### 3. 答案:C
解析:OpenAI function calling 协议要求每个 tool_call 都要有对应的 ToolMessage 响应。返回拒绝 ToolMessage(带正确的 `tool_call_id`)能让 LLM 理解"工具被拒",从而重新组织回答。不返回会导致 LLM 困惑或报错。

### 4. 答案:`langgraph.types`
解析:`from langgraph.types import interrupt, Command`。

### 5. 答案:`tasks`
解析:`state.tasks` 是一个列表,每个 task 可能有 `interrupts` 属性。若 `task.interrupts` 非空,说明在该 task 处暂停。`task.interrupts[0].value` 是传给 `interrupt()` 的参数。

### 6. 答案:`Checkpointer`
解析:暂停时状态必须持久化才能恢复。没有 Checkpointer,interrupt 无法工作(编译时会报错)。

### 7. 参考代码

```python
"""Day5 练习题7:可配置阈值的转账审批"""
import os
import sqlite3
from typing import TypedDict, Annotated
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, ToolMessage
from langgraph.graph import StateGraph, START, END
from langgraph.graph.message import add_messages
from langgraph.checkpoint.sqlite import SqliteSaver
from langgraph.types import interrupt, Command

load_dotenv()


def transfer(to: str, amount: float) -> str:
    return f"已向 {to} 转账 {amount} 元"


tools = [transfer]
tools_by_name = {t.__name__: t for t in tools}
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0).bind_tools(tools)


class AgentState(TypedDict):
    messages: Annotated[list, add_messages]
    approval_threshold: float


def agent_node(state):
    return {"messages": [llm.invoke(state["messages"])]}


def smart_tool_node(state):
    last: AIMessage = state["messages"][-1]
    threshold = state.get("approval_threshold", 1000)
    out = []
    for c in last.tool_calls:
        if c["name"] == "transfer":
            amount = float(c["args"].get("amount", 0))
            if amount > threshold:
                decision = interrupt({
                    "type": "transfer_approval",
                    "to": c["args"]["to"],
                    "amount": amount,
                    "threshold": threshold,
                })
                if str(decision).strip().lower() not in ("yes", "ok", "y", "允许"):
                    out.append(ToolMessage(
                        content=f"人工拒绝转账 {amount} 元",
                        tool_call_id=c["id"],
                    ))
                    continue
        try:
            r = tools_by_name[c["name"]](**c["args"])
        except (KeyError, TypeError, ValueError) as e:
            r = f"错误:{type(e).__name__}"
        out.append(ToolMessage(content=str(r), tool_call_id=c["id"]))
    return {"messages": out}


def should_use_tool(state):
    last = state["messages"][-1]
    if hasattr(last, "tool_calls") and last.tool_calls:
        return "tools"
    return END


builder = StateGraph(AgentState)
builder.add_node("agent", agent_node)
builder.add_node("tools", smart_tool_node)
builder.add_edge(START, "agent")
builder.add_conditional_edges("agent", should_use_tool)
builder.add_edge("tools", "agent")

conn = sqlite3.connect("threshold.db", check_same_thread=False)
graph = builder.compile(checkpointer=SqliteSaver(conn))


def test(threshold: float, amount: float):
    config = {"configurable": {"thread_id": f"t-{threshold}-{amount}"}}
    graph.invoke(
        {
            "messages": [HumanMessage(content=f"给 alice 转 {amount} 元")],
            "approval_threshold": threshold,
        },
        config=config,
    )
    state = graph.get_state(config)
    if state.next:
        # 暂停了 → 需要审批
        print(f"阈值 {threshold}, 转账 {amount}:已暂停待审批")
        # 自动批准
        graph.invoke(Command(resume="yes"), config=config)
        state = graph.get_state(config)
        print("  → 批准后:", state.values["messages"][-1].content)
    else:
        print(f"阈值 {threshold}, 转账 {amount}:未暂停,直接执行")
        print("  → 结果:", state.values["messages"][-1].content)


if __name__ == "__main__":
    test(500, 800)    # 800 > 500,应审批
    test(1000, 800)   # 800 < 1000,应直接执行
```

```python
"""Day5 练习题8:代码生成审批图"""
import os
import sqlite3
from typing import TypedDict
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.sqlite import SqliteSaver
from langgraph.types import interrupt, Command

load_dotenv()


class CodeState(TypedDict):
    task: str
    code: str
    approved_code: str


def coder(state: CodeState) -> dict:
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
    prompt = f"用 Python 实现以下任务,只输出代码,不要解释:{state['task']}"
    code = llm.invoke(prompt).content
    return {"code": code}


def reviewer(state: CodeState) -> Command:
    feedback = interrupt({
        "prompt": "请审核代码",
        "code": state["code"],
        "hint": "输入 'ok' 通过,或输入修改后的完整代码",
    })
    if str(feedback).strip().lower() in ("ok", "通过", "approve"):
        return Command(update={"approved_code": state["code"]}, goto="run")
    return Command(update={"approved_code": str(feedback).strip()}, goto="run")


def run(state: CodeState) -> dict:
    print("代码已批准:")
    print(state["approved_code"])
    return {"approved_code": state["approved_code"]}


builder = StateGraph(CodeState)
builder.add_node("coder", coder)
builder.add_node("reviewer", reviewer)
builder.add_node("run", run)
builder.add_edge(START, "coder")
builder.add_edge("coder", "reviewer")
builder.add_edge("run", END)

conn = sqlite3.connect("code_review.db", check_same_thread=False)
graph = builder.compile(checkpointer=SqliteSaver(conn))


def main():
    config = {"configurable": {"thread_id": "code-1"}}
    graph.invoke({"task": "写一个函数计算斐波那契数列第 n 项", "code": "", "approved_code": ""}, config=config)
    state = graph.get_state(config)
    print("=== 生成的代码 ===")
    print(state.values["code"])
    human = input("你的审核意见(ok 或修改稿): ")
    result = graph.invoke(Command(resume=human), config=config)
    print("\n=== 最终批准的代码 ===")
    print(result["approved_code"])


if __name__ == "__main__":
    main()
```
