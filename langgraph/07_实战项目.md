> 今天我们把前 6 天所学融会贯通,从零设计并实现一个完整的多 Agent 智能客服系统。它包含知识库检索(RAG)、多 Agent 协作(Supervisor 模式)、持久化记忆、人机协同(转人工审批)。

1. 综合运用 StateGraph、Reducer、条件边、Command、Checkpointer、interrupt 等全部核心能力。
2. 独立设计一个真实可用的多 Agent 系统架构,理解各模块职责。
3. 实现一个"智能客服系统":RAG 问答 + 工单创建 + 人工转接审批 + 多轮记忆。

## 二、项目背景与架构设计

### 2.1 场景

我们要为一家手机厂商构建智能客服系统,需求:
- 用户提问产品问题 → 系统从知识库检索答案(RAG)。
- 检索不到或用户不满意 → 创建工单并尝试自动回复。
- 涉及退款/投诉等敏感操作 → 转人工,需人工审批后执行。
- 多轮对话有记忆,用户离开再回来能续上。

### 2.2 架构总览

```
用户消息
   ↓
Supervisor(主管 Agent)
   ↓ (路由)
   ├──> rag_agent        (知识库问答)
   ├──> ticket_agent     (创建工单)
   └──> human_agent      (转人工,需审批)
   ↑___________________________________|
   (各 Agent 完成后回 Supervisor)
   ↓
Supervisor 判断 FINISH → END
```

### 4.1 第一步:知识库与 RAG 模块

```python
"""Day7 实战 - 知识库与 RAG 检索(关键词模拟版)"""
from typing import List, Dict


# 模拟知识库:生产环境替换为向量库
KNOWLEDGE_BASE: List[Dict] = [
    {
        "id": "kb001",
        "product": "Phone X1",
        "question": "电池续航",
        "answer": "Phone X1 配备 5000mAh 电池,日常使用续航约 1.5 天,支持 66W 快充。",
    },
    {
        "id": "kb002",
        "product": "Phone X1",
        "question": "屏幕尺寸",
        "answer": "Phone X1 采用 6.7 英寸 OLED 屏幕,分辨率 2772x1344,支持 120Hz 刷新率。",
    },
    {
        "id": "kb003",
        "product": "Phone X1",
        "question": "保修政策",
        "answer": "Phone X1 整机保修 1 年,电池保修 1 年,屏幕非人为损坏 6 个月内免费更换。",
    },
    {
        "id": "kb004",
        "product": "Phone X2",
        "question": "电池续航",
        "answer": "Phone X2 配备 6000mAh 电池,日常使用续航约 2 天,支持 100W 快充。",
    },
]


def retrieve(query: str, top_k: int = 2) -> List[Dict]:
    """简易检索:按关键词重叠度排序。生产环境换成向量检索。"""
    scored = []
    query_words = set(query.lower().replace("?", "").replace("?", "").split())
    for doc in KNOWLEDGE_BASE:
        doc_text = (doc["product"] + " " + doc["question"] + " " + doc["answer"]).lower()
        overlap = sum(1 for w in query_words if w in doc_text)
        scored.append((overlap, doc))
    scored.sort(key=lambda x: x[0], reverse=True)
    # 只返回有重叠的
    return [doc for score, doc in scored[:top_k] if score > 0]


def rag_answer(query: str) -> Dict:
    """检索 + 生成答案"""
    docs = retrieve(query)
    if not docs:
        return {"found": False, "answer": "", "sources": []}
    # 简化:直接拼答案。生产环境用 LLM 综合
    answer = "\n".join([d["answer"] for d in docs])
    return {"found": True, "answer": answer, "sources": [d["id"] for d in docs]}


if __name__ == "__main__":
    # 测试 RAG
    print(rag_answer("Phone X1 电池续航怎么样"))
    print(rag_answer("Phone X2 屏幕"))  # 没有屏幕数据,found=False
```

### 4.3 第三步:Supervisor 节点

```python
"""Day7 实战 - Supervisor 节点"""
import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage
from langgraph.types import Command

load_dotenv()
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)


def supervisor(state: CustomerServiceState) -> Command:
    """主管:决定下一步派谁"""
    decision_llm = llm.bind_tools([{
        "name": "route",
        "description": "选择下一个 Agent",
        "parameters": {
            "type": "object",
            "properties": {
                "next": {
                    "type": "string",
                    "enum": ["rag_agent", "ticket_agent", "human_agent", "FINISH"],
                },
                "reason": {"type": "string"},
            },
            "required": ["next"],
        },
    }], tool_choice="route")

    history_tail = "\n".join(
        [f"{m.type}: {m.content[:100]}" for m in state["messages"][-6:]]
    )
    prompt = f"""你是客服系统主管。根据对话历史决定下一步。

对话历史(最近6条):
{history_tail}

当前查询:{state.get('query', '')}
已有 RAG 答案:{"有" if state.get('rag_result') else "无"}
已有工单:{"有" if state.get('ticket_id') else "无"}
已标记转人工:{"是" if state.get('needs_human') else "否"}

决策规则:
1. 如果当前查询还没回答过,且没有 RAG 答案 → rag_agent
2. 如果 RAG 答案已有但用户不满意(含"退款/投诉/转人工"等)→ human_agent
3. 如果需要创建工单(用户问题未解决)且还没工单 → ticket_agent
4. 如果已经回答完毕且无需工单 → FINISH
5. 如果已转人工且已处理 → FINISH
"""
    resp = decision_llm.invoke([HumanMessage(content=prompt)])
    nxt = resp.tool_calls[0]["args"]["next"]

    if nxt == "FINISH":
        return Command(
            update={"messages": [AIMessage(content="(主管)任务完成,结束本轮")]},
            goto="__end__",
        )
    return Command(update={"next": nxt}, goto=nxt)
```

### 4.4 第四步:三个工人 Agent

```python
"""Day7 实战 - 三个工人 Agent"""
from langchain_core.messages import AIMessage


# === RAG Agent ===
def rag_agent(state: CustomerServiceState) -> dict:
    """检索知识库回答"""
    query = state["query"]
    result = rag_answer(query)
    if result["found"]:
        msg = f"[RAG 回答] {result['answer']}\n(来源:{', '.join(result['sources'])})"
        return {
            "rag_result": result["answer"],
            "messages": [AIMessage(content=msg)],
        }
    # 没找到,标记可能需要人工
    return {
        "rag_result": "",
        "needs_human": True,
        "messages": [AIMessage(content="[RAG] 抱歉,知识库中未找到相关答案,可能需要转人工。")],
    }


# === Ticket Agent ===
import time


def ticket_agent(state: CustomerServiceState) -> dict:
    """创建工单"""
    ticket_id = f"TK-{int(time.time()) % 100000}"
    return {
        "ticket_id": ticket_id,
        "messages": [AIMessage(content=f"[工单] 已为您创建工单 {ticket_id},客服将在 24 小时内联系您。")],
    }


# === Human Agent(带 interrupt 审批)===
from langgraph.types import interrupt


def human_agent(state: CustomerServiceState) -> dict:
    """转人工,需人工审批"""
    query = state["query"]
    decision = interrupt({
        "type": "human_handoff_approval",
        "user_id": state.get("user_id", "unknown"),
        "query": query,
        "message": f"用户问题需转人工:{query}",
    })
    if str(decision).strip().lower() in ("approve", "yes", "ok", "同意"):
        return {
            "needs_human": False,  # 已处理
            "messages": [AIMessage(content=f"[人工] 已转接人工客服,工号 {decision} 为您服务。")],
        }
    return {
        "needs_human": False,
        "messages": [AIMessage(content="[人工] 人工转接被拒绝,建议您稍后重试或创建工单。")],
    }
```

```python
"""Day7 实战 - 组装完整图"""
import sqlite3
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.sqlite import SqliteSaver


def build_graph(db_path: str = "customer_service.db"):
    builder = StateGraph(CustomerServiceState)
    builder.add_node("supervisor", supervisor)
    builder.add_node("rag_agent", rag_agent)
    builder.add_node("ticket_agent", ticket_agent)
    builder.add_node("human_agent", human_agent)

    builder.add_edge(START, "supervisor")
    builder.add_edge("rag_agent", "supervisor")
    builder.add_edge("ticket_agent", "supervisor")
    builder.add_edge("human_agent", "supervisor")
    # supervisor 用 Command 路由,无需显式 conditional_edges

    conn = sqlite3.connect(db_path, check_same_thread=False)
    return builder.compile(checkpointer=SqliteSaver(conn))
```

### 4.6 第六步:运行入口(处理 interrupt 循环)

```python
"""Day7 实战 - 运行入口"""
from langgraph.types import Command


def chat(graph, thread_id: str, user_input: str, auto_approve_human: bool = True) -> str:
    """与客服系统对话,自动处理 interrupt"""
    config = {"configurable": {"thread_id": thread_id}}
    graph.invoke(
        {
            "messages": [HumanMessage(content=user_input)],
            "user_id": thread_id,
            "query": user_input,
            "rag_result": "",
            "ticket_id": "",
            "needs_human": False,
            "next": "",
        },
        config=config,
    )

    # 处理可能的 interrupt(human_agent 审批)
    while True:
        state = graph.get_state(config)
        if not state.next:
            break
        # 检查 interrupt
        interrupted = False
        tasks = state.tasks or []
        for task in tasks:
            if hasattr(task, "interrupts") and task.interrupts:
                info = task.interrupts[0].value
                print(f"\n[审批请求] {info.get('message', '')}")
                if auto_approve_human:
                    print(">>> 自动批准转人工")
                    graph.invoke(Command(resume="approve"), config=config)
                else:
                    ans = input("是否批准?(yes/no): ")
                    graph.invoke(
                        Command(resume="approve" if ans.lower() in ("yes", "y") else "reject"),
                        config=config,
                    )
                interrupted = True
                break
        if not interrupted:
            graph.invoke(None, config=config)

    final = graph.get_state(config)
    return final.values["messages"][-1].content
```

```python
]

    user_id: str
    query: str
    rag_result: str
    ticket_id: str
    needs_human: bool

# ============ 3. LLM 与 Supervisor ============

def supervisor(state: CustomerServiceState) -> Command:
    decision_llm = llm.bind_tools([{

            "properties": {
                "next": {"type": "string",
                         "enum": ["rag_agent", "ticket_agent", "human_agent", "FINISH"]},

    history_tail = "\n".join(

    sensitive_keywords = ["退款", "投诉", "转人工", "赔偿"]

    conn = sqlite3.connect(db_path, check_same_thread=False)
    return builder.compile(checkpointer=SqliteSaver(conn))

def chat(graph, thread_id: str, user_input: str, auto_approve: bool = True) -> str:
    config = {"configurable": {"thread_id": thread_id}}
    graph.invoke(

        if not state.next:
```

## 五、项目扩展方向

这个项目是骨架,可朝以下方向扩展:

### 1. 答案:B

### 2. 答案:B

```python
"""Day7 练习题8:用 Chroma 向量检索替代关键词匹配"""
# pip install chromadb langchain-chroma
import os
from langchain_chroma import Chroma
from langchain_openai import OpenAIEmbeddings
from langchain_core.documents import Document

load_dotenv()

# 1. 初始化 embeddings
embeddings = OpenAIEmbeddings(model="text-embedding-3-small")

# 2. 把知识库转成 Document 并灌入 Chroma
docs = [
    Document(page_content=d["product"] + " " + d["question"] + " " + d["answer"],
             metadata={"id": d["id"], "product": d["product"]})
    for d in KNOWLEDGE_BASE
]
# 持久化到本地目录
vectorstore = Chroma.from_documents(docs, embeddings, persist_directory="./chroma_db")

# 3. 替换 retrieve 函数
def retrieve(query: str, top_k: int = 2) -> List[Dict]:
    results = vectorstore.similarity_search_with_relevance_scores(query, k=top_k)
    out = []
    for doc, score in results:
        if score > 0.3:  # 相似度阈值
            # 从 metadata 找回原始知识库条目
            kid = doc.metadata["id"]
            original = next((d for d in KNOWLEDGE_BASE if d["id"] == kid), None)
            if original:
                out.append(original)
    return out

# 其余代码(rag_answer 等)不变,因为 retrieve 接口一致
# 测试:
# print(rag_answer("Phone X1 电池能用多久"))  # 语义匹配,即使没出现"续航"二字也能命中
```

关键改造点:
- `retrieve` 的函数签名和返回类型保持不变,上层代码(`rag_answer`)无需修改。
- 用 `similarity_search_with_relevance_scores` 做语义相似度检索,带阈值过滤。
- `persist_directory` 让向量库持久化到磁盘,下次可直接 `Chroma(persist_directory=...)` 加载。
- 语义检索的优势:用户问"电池能用多久"也能匹配到"电池续航"的知识。
