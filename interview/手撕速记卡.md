#### 1. Agent Loop

```python
# 三重安全网: max_iter + token_budget + llm_timeout(缺一不可)
for _ in range(max_iterations):
    if total_tokens >= token_budget:                  # 安全网1: 预算
        return summarize(messages)                    # 软停止: 让 LLM 收尾而非硬截断
    response = llm_with_timeout(messages, tool_schema) # 安全网2: 单次调用超时
    total_tokens += usage(response)

    messages.append(assistant_msg(response))
    if not response["tool_calls"]:                     # 退出条件: 无工具调用
        return response["content"]

    for tc in response["tool_calls"]:                  # 工具: 未知工具/异常/超时 3 类都要兜住
        result = execute_tool_with_retry(tc)           # 重试有上限(如 2 次)
        messages.append({"role": "tool", "name": tc["name"], "content": result})

return f"[STOP] 达到最大迭代 {max_iterations}"           # 安全网3: 迭代上限
```

#### 2. ReAct Agent

```python
# 关键: Final Answer 优先解析 + 正则容错 + 死循环检测
for step in range(max_steps):
    raw = llm(prompt)

    parsed = parse(raw)                     # 正则: Final Answer -> Thought -> Action -> Action Input
    if parsed["final_answer"]:              # 必须先判 Final Answer,否则会被误判成 Action
        return parsed["final_answer"]

    action, action_input = parsed["action"], parsed["action_input"]
    if action not in actions:               # 无效工具名 → 不静默失败
        return f"[ERROR] 无效 Action: {action}"
    if detect_loop(history, action, action_input):  # 连续 3 次相同 (action, input)
        return f"[STOP] 检测到死循环: {action}"

    try:
        observation = actions[action](action_input)
    except Exception as e:
        observation = f"[ERROR] {e}"        # 异常转成 Observation 回灌,让 LLM 自我纠正
    history.append((action, action_input))
    prompt += f"\nThought: {parsed['thought']}\nAction: {action}\nAction Input: {action_input}\nObservation: {observation}\n"
    trace.append({...})                     # 每步落 trace: thought/action/observation
```

#### 4. 多策略分块器

```python
# 策略模式: 统一接口,4 种策略可插拔
class ChunkStrategy(Protocol):
    def chunk(self, text: str) -> list[str]: ...

FixedSizeChunker:                        # 定长 + overlap: step = chunk_size - overlap
    while start < len(text):
        chunks.append(text[start:start + chunk_size])
        start += chunk_size - overlap

RecursiveChunker:                        # 按分隔符层级递归,逐级降级
    separators = ["\n\n", "\n", "。", ".", " ", ""]
    if len(text) <= chunk_size or not separators: return [text]
    sep, rest = separators[0], separators[1:]
    for part in text.split(sep):
        if len(part) <= chunk_size: chunks.append(part)      # 放得下 → 成块
        else: chunks += recurse(part, rest)                  # 放不下 → 降级递归
    # sep == "" 时兜底硬切

SemanticChunker:                         # 相邻句子相似度 < 阈值则断开(超 max_size 也断)
    for s in sentences[1:]:
        if sim(current[-1], s) >= threshold and len(cur) < max_size: cur.append(s)
        else: flush(cur); cur = [s]       # < min_size 的块回并到上一块

MarkdownChunker:                         # 按 r"^#{1,6}\s" 切,标题留在块首
    for line in lines:
        if re.match(r"^#{1,6}\s", line): flush(current_section)
        current_section.append(line)
    # 单块仍超长 → 交给 RecursiveChunker 二次切
```

#### 5. 混合检索

```python
# 关键: 两路召回取大 k,RRF 只用排名,精排放在融合之后
bm25_hits = BM25.search(query, k=10)          # 稀疏: TF-IDF + 文档长度归一化(k1=1.5, b=0.75)
vec_hits  = Vector.search(query, k=10)        # 稠密: 余弦相似度,doc embedding 预计算

def rrf(result_lists, k=60):                  # score(d) = Σ 1/(k + rank(d))
    scores = defaultdict(float)
    for hits in result_lists:                 # 只看排名,不看分数 → 天然归一化
        for rank, (doc_id, _) in enumerate(hits, 1):
            scores[doc_id] += 1 / (k + rank)  # k 平滑: 越大越平,60 为经验值
    return sorted(scores.items(), key=lambda x: x[1], reverse=True)

fused    = rrf([bm25_hits, vec_hits])[:top_k * 2]    # 先粗融合缩小候选集
reranked = Reranker.rerank(query, docs, [d for d, _ in fused])[:top_k]  # 再 cross-encoder 精排
```

#### 6. Agentic RAG

```python
# 关键: 路由 → 检索 → 评估 → (不足则改写重检) → 生成
strategy = route(question)                  # keyword / semantic / graph
```

```python
messages = [system_msg, retrieval_msg, *recent_history, user_msg]

def compress_history(history, target):      # 三段式: 首轮保留 + 中间摘要 + 最近 N 轮
    first = history[0]
    recent = 末尾回填直到 target * 0.6
    middle = history[1:len(history) - len(recent)]
    return [first, {"role": "system", "content": llm_summarize(middle)}, *recent]
```

#### 8. Multi-Agent(Supervisor)

```python
# 关键: 依赖就绪才执行 + 无就绪即死锁 + 结果按 task_id 聚合
subtasks = supervisor.decompose(task)        # [{"id", "type", "desc", "worker", "depends_on"}]
results, done = {}, set()

while len(done) < len(subtasks):
    ready = [st for st in subtasks             # 依赖全部完成 → 本轮就绪
             if st["id"] not in done and all(d in done for d in st.get("depends_on", []))]
    if not ready:                              # 有未完成却无就绪 → 环形依赖,报死锁而非 hang
        return f"[WARN] 死锁,剩余 {[st['id'] for st in subtasks if st['id'] not in done]}"
    with ThreadPoolExecutor(max_workers) as ex:  # 就绪任务并行,层间天然串行
        for st in ready:
            inputs = {deps_type: results[d]["result"] for d in st.get("depends_on", [])}  # 上游输出作输入
            futures[ex.submit(workers[st["worker"]].execute, st, inputs)] = st
        for future in as_completed(futures):
            st = futures[future]
            results[st["id"]] = future.result(); done.add(st["id"])  # worker 失败重试后降级返回

return aggregate(results) if success_ratio >= 0.6 else "[FAILED] 需人工介入"  # 质量门禁
```

#### 常错速查

| #   | 易错点                       | 正确做法                                     |
| --- | ---------------------------- | -------------------------------------------- |
| 1   | `while True:` 裸循环         | max_iter + token_budget + timeout 三重安全网 |
| 2   | 工具裸调用 `tool(**args)`    | try/except 包裹,异常转 error 消息回灌 LLM    |
| 3   | ReAct 先解析 Action          | Final Answer 优先级更高,先判它               |
| 4   | `results.append(f.result())` | 按下标/`call_id` 回填,否则乱序               |
| 5   | RRF 里加原始分数             | 只用排名 `1/(k + rank)`,分数量纲不同不能相加 |
| 6   | 压缩丢掉首轮                 | 首轮保留 + 中间摘要 + 最近 N 轮完整          |
| 7   | 固定窗口限流                 | 滑动窗口(deque + 时间戳),避免边界突刺        |
| 8   | 依赖不满足就死等             | 无就绪任务但有未完成 → 报死锁                |
| 9   | span 全平铺                  | 栈维护 current_span,新 span 的 parent 是栈顶 |
| 10  | `except: pass`               | `except Exception as e: log(e)`              |

#### 附：题型索引与答题套路

**答题套路（动手写代码前先说这 4 句）**

1. 说清输入输出与边界（空输入 / 超长 / 并发 / 超时）
2. 先给暴力解，说明时间与空间复杂度
3. 说优化方向（用排名代替分数、用缓存换延迟、用并发换吞吐）
4. 写完补边界处理 + 自测用例

**题型索引**

| # | 题目 | 面试频率 | 常挂点 |
| --- | --- | --- | --- |
| 1 | Agent Loop | 极高 | 缺 `token_budget` / 超时，写成 `while True` |
| 2 | ReAct Agent | 极高 | 先解析 Action，漏掉 Final Answer |
| 3 | 并行工具调用 | 高 | 用 `append` 导致结果与 `tool_call` 乱序 |
| 4 | 多策略分块器 | 中 | 递归缺少终止条件 |
| 5 | 混合检索 + RRF | 极高 | 把原始分数也加进 RRF |
| 6 | Agentic RAG | 高 | 无 `max_iter`，无限检索 |
| 7 | 上下文管理器 | 高 | 压缩丢掉首轮 / 没给输出留额度 |
| 8 | Multi-Agent Supervisor | 高 | 环形依赖死等，而非报死锁 |
| 9 | 安全防护层 | 中 | 只做输入过滤，漏掉工具越权 |
| 10 | 可观测性 | 中 | span 全平铺，没有父子关系 |
